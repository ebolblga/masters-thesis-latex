\section*{Перечень сокращений и специальных терминов}

\addcontentsline{toc}{section}{ПЕРЕЧЕНЬ СОКРАЩЕНИЙ И СПЕЦИАЛЬНЫХ ТЕРМИНОВ}

\textbf{Сокращения}

\begin{description}
	\item[API] (Application Programming Interface) --- программный интерфейс приложения
	\item[ERD] (Entity-Relationship Diagram) --- модель данных, позволяющая описывать концептуальные схемы предметной области
	\item[ERP] (Enterprise Resource Planning) --- планирование ресурсов предприятия
	\item[PWA] (Progressive Web Application) --- прогрессивное веб-приложение
	\item[REST] (Representational State Transfer) --- архитектурный стиль взаимодействия компонентов распределённого приложения в сети
	\item[SA] (Simulated Annealing) --- имитация отжига
	\item[SAT] (Boolean Satisfiability Problem) --- задача булевой выполнимости
	\item[SMT] (Satisfiability Modulo Theories) --- выполнимость по моделям теорий
	\item[UI] (User Interface) --- пользовательский интерфейс
	\item[WFC] (Wave Function Collapse) --- коллапс волновой функции
	\item[СУБД] (Система управления базами данных)
	\item[УМУ] (Учебно-методическое управление)
	\item[ЭОС] (Электронная образовательная среда)
\end{description}

\textbf{Термины}

\begin{description}
	\item[Парсинг] --- автоматизированный сбор и структурирование информации
	\item[Фреймворк] структурированный набор инструментов и библиотек, облегчающий разработку программного обеспечения
\end{description}
